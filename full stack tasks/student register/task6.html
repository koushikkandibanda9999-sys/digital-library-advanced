About

Automated logging records database actions like INSERT and UPDATE automatically. It is used for tracking, auditing, and maintaining history.

Database Tables

Main Table

CREATE TABLE employees (
 emp_id INT PRIMARY KEY,
 name VARCHAR(50),
 salary INT
);

Log Table

CREATE TABLE employee_log (
 log_id INT AUTO_INCREMENT PRIMARY KEY,
 emp_id INT,
 action_type VARCHAR(10),
 action_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Triggers







    
Daily Activity View

    
CREATE VIEW daily_activity AS
SELECT DATE(action_time) AS activity_date,
       action_type,
       COUNT(*) AS total_actions
FROM employee_log
GROUP BY DATE(action_time), action_type;
